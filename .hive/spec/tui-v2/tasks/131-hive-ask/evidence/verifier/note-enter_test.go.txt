package tui
import("fmt";"testing";tea "charm.land/bubbletea/v2";"github.com/Harrison-Blair/hive/internal/ask")
func TestVerifierNoteEnterPicksCurrentOption(t *testing.T){
 m,calls:=askModel(t,120,40,oneQuestionAsk("a1","manager-hive"))
 m,_=raise(t,m,tea.KeyPressMsg{Code:tea.KeyDown})
 m,_=raise(t,m,tea.KeyPressMsg{Code:tea.KeyTab})
 m,_=raise(t,m,key('o',"o"));m,_=raise(t,m,key('k',"k"))
 run(t,m,tea.KeyPressMsg{Code:tea.KeyEnter})
 want:=[]askCall{{"answer","a1",[]ask.Reply{{Picks:[]int{2},Note:"ok"}}}}
 if fmt.Sprint(*calls)!=fmt.Sprint(want){t.Fatalf("note enter: got %v, want %v",*calls,want)}
}
