import json,time,copy
from live import cli,tmp,ev
q={'question':'Q?','header':'H','options':[{'label':'A','description':'alpha'},{'label':'B','description':'beta'}]}
cases=[('not JSON','{','ask JSON'),('unknown field',{'questions':[q],'extra':1},'unknown field'),('no question',{'questions':[]},'1 to 4 questions'),('five questions',{'questions':[q]*5},'1 to 4 questions')]
for name,key,value,want in [('header length','header','thirteen char','header of 12'),('empty header','header','','header'),('one option','options',q['options'][:1],'2 to 5 options'),('six options','options',q['options']*3,'2 to 5 options'),('empty question','question','','question text')]:
 question=copy.deepcopy(q);question[key]=value;cases.append((name,{'questions':[question]},want))
for name,key,want in [('empty label','label','label'),('empty description','description','description')]:
 question=copy.deepcopy(q);question['options'][0][key]='';cases.append((name,{'questions':[question]},want))
state=tmp/'home/.hive/state/asks';before=sorted(x.name for x in state.glob('*.json'));results=[]
for name,data,want in cases:
 (tmp/'ask1.json').write_text(data if isinstance(data,str) else json.dumps(data));(tmp/'ask1.out').unlink(missing_ok=True)
 cli('pane','send-text','w1:p2','make-ask-1');cli('pane','send-keys','w1:p2','enter')
 for _ in range(100):
  if (tmp/'ask1.out').exists() and 'elapsed_ns=' in (tmp/'ask1.out').read_text():break
  time.sleep(.1)
 output=(tmp/'ask1.out').read_text();after=sorted(x.name for x in state.glob('*.json'))
 assert 'exit=1' in output and want in output and after==before,(name,output,after)
 results.append({'case':name,'output':output,'records':after,'result':'pass'})
(ev/'invalid-cli.json').write_text(json.dumps(results,indent=2)+'\n')
