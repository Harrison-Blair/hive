package main

import (
 "fmt"
 "os"
 "strconv"
 "strings"
 "github.com/Harrison-Blair/hive/internal/lib/cellgrid"
)
func must(err error) {if err != nil {panic(err)}}
func load(path string) cellgrid.Grid {f,e:=os.Open(path);must(e);defer f.Close();g,e:=cellgrid.ReadGrid(f);must(e);return g}
func main(){
 a:=os.Args; want:=load(a[2]); var got cellgrid.Grid
 if strings.HasSuffix(a[1],".grid") {got=load(a[1])} else {
 data,e:=os.ReadFile(a[1]);must(e);pw,ph:=want.Width,want.Height
 if len(a)>5 {pw,e=strconv.Atoi(a[4]);must(e);ph,e=strconv.Atoi(a[5]);must(e)}
 full,e:=cellgrid.Convert(string(data),pw,ph);must(e)
 got=cellgrid.Grid{Width:want.Width,Height:want.Height,Cells:make([]cellgrid.Cell,want.Width*want.Height)}
 for r:=0;r<want.Height;r++ {copy(got.Cells[r*want.Width:(r+1)*want.Width],full.Cells[r*pw:r*pw+want.Width])}
 }
 diffs,e:=cellgrid.Compare(want,got,nil);must(e)
 must(os.WriteFile(a[3],[]byte(cellgrid.FormatDiffs(diffs)),0644))
 fmt.Printf("%s: %d cells differ of %d\n",a[1],len(diffs),len(want.Cells))
}
