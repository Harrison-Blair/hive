```go
// Command griddiff lists the changed cells of each grid in two folders.
package main

import (
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"strings"

	"github.com/Harrison-Blair/hive/internal/lib/cellgrid"
)

func read(p string) cellgrid.Grid {
	f, err := os.Open(p)
	if err != nil {
		panic(err)
	}
	defer f.Close()
	g, err := cellgrid.ReadGrid(f)
	if err != nil {
		panic(err)
	}
	return g
}

func desc(c cellgrid.Cell) string {
	b := ""
	if c.Bold {
		b = "+b"
	}
	return fmt.Sprintf("%s%s", c.Token, b)
}

func main() {
	olds, _ := filepath.Glob(filepath.Join(os.Args[1], "*.grid"))
	sort.Strings(olds)
	total := 0
	for _, o := range olds {
		name := filepath.Base(o)
		diffs, err := cellgrid.Compare(read(o), read(filepath.Join(os.Args[2], name)), nil)
		if err != nil {
			panic(err)
		}
		fmt.Printf("%s: %d cells\n", name, len(diffs))
		total += len(diffs)
		// Group runs of adjacent columns in one row.
		for i := 0; i < len(diffs); {
			j := i
			for j+1 < len(diffs) && diffs[j+1].Row == diffs[i].Row && diffs[j+1].Col == diffs[j].Col+1 {
				j++
			}
			var ow, nw, ot, nt []string
			for _, d := range diffs[i : j+1] {
				ow = append(ow, string(d.Want.Rune))
				nw = append(nw, string(d.Got.Rune))
				ot = append(ot, desc(d.Want))
				nt = append(nt, desc(d.Got))
			}
			fmt.Printf("  row %d cols %d-%d: %q [%s] -> %q [%s]\n", diffs[i].Row, diffs[i].Col, diffs[j].Col,
				strings.Join(ow, ""), uniq(ot), strings.Join(nw, ""), uniq(nt))
			i = j + 1
		}
	}
	fmt.Printf("total: %d cells\n", total)
}

func uniq(s []string) string {
	seen := map[string]bool{}
	var out []string
	for _, x := range s {
		if !seen[x] {
			seen[x] = true
			out = append(out, x)
		}
	}
	return strings.Join(out, ",")
}
```
