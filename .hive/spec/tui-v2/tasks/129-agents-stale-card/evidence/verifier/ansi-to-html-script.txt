import html
import re
from pathlib import Path

root = Path('.hive/spec/tui-v2/tasks/129-agents-stale-card/evidence/verifier')
out = Path('.hive/tmp/t129-verifier-firefox')
sgr = re.compile(r'\x1b\[([0-9;]*)m')
for source in root.glob('live-*.ansi'):
    fg, bg, bold = '#e6e1d6', '', False
    rows = []
    for line in source.read_text().splitlines():
        cells = []
        position = 0
        for match in list(sgr.finditer(line)) + [None]:
            end = match.start() if match else len(line)
            text = line[position:end]
            style = f'color:{fg};font-weight:{700 if bold else 400};'
            if bg:
                style += f'background:{bg};'
            # Each character has one terminal cell, as in the approved boards.
            cells.extend(f'<span style="{style}">{html.escape(char)}</span>' for char in text)
            if match is None:
                break
            codes = [int(code) if code else 0 for code in match.group(1).split(';')]
            i = 0
            while i < len(codes):
                code = codes[i]
                if code == 0:
                    fg, bg, bold = '#e6e1d6', '', False
                elif code == 1:
                    bold = True
                elif code == 22:
                    bold = False
                elif code == 39:
                    fg = '#e6e1d6'
                elif code == 49:
                    bg = ''
                elif code in (38, 48) and codes[i+1] == 2:
                    colour = '#%02x%02x%02x' % tuple(codes[i+2:i+5])
                    if code == 38:
                        fg = colour
                    else:
                        bg = colour
                    i += 4
                else:
                    raise ValueError((source, code, codes))
                i += 1
            position = match.end()
        rows.append('<div>' + ''.join(cells) + '</div>')
    document = '''<!doctype html><meta charset="utf-8"><style>
body{margin:0}pre{margin:0;box-sizing:border-box;width:984px;height:728px;padding:24px;background:#171612;border-radius:10px;font-family:'JetBrains Mono',monospace;font-size:13px;line-height:17px;white-space:pre;color:#e6e1d6;overflow:hidden;font-variant-ligatures:none}pre div{height:17px}span{display:inline-block;vertical-align:top;width:1ch;overflow:hidden}
</style><pre>''' + ''.join(rows) + '</pre>'
    (out/(source.stem+'.html')).write_text(document)
    print(source.name, len(rows), 'rows')
