// Command compare crops the top-left W×H cells of a pane capture, converts
// them with cellgrid, and compares them with a board grid under the live
// rule: in a space cell of the capture, the token and the bold flag are not
// compared. Scratch tool of task 38-screen-friction.
package main

import (
	"fmt"
	"os"
	"strconv"
	"strings"

	"github.com/Harrison-Blair/hive/internal/lib/cellgrid"
	"github.com/charmbracelet/x/ansi"
)

func main() {
	gridFile, captureFile := os.Args[1], os.Args[2]
	width, _ := strconv.Atoi(os.Args[3])
	height, _ := strconv.Atoi(os.Args[4])
	f, err := os.Open(gridFile)
	if err != nil {
		panic(err)
	}
	want, err := cellgrid.ReadGrid(f)
	if err != nil {
		panic(err)
	}
	raw, err := os.ReadFile(captureFile)
	if err != nil {
		panic(err)
	}
	lines := strings.Split(strings.ReplaceAll(string(raw), "\r\n", "\n"), "\n")
	if len(lines) < height {
		panic(fmt.Sprintf("capture has %d lines, want %d", len(lines), height))
	}
	for i := range lines[:height] {
		lines[i] = ansi.Truncate(lines[i], width, "") + "\x1b[0m"
	}
	got, err := cellgrid.Convert(strings.Join(lines[:height], "\n"), width, height)
	if err != nil {
		panic(err)
	}
	diffs, err := cellgrid.Compare(want, got, nil)
	if err != nil {
		panic(err)
	}
	var live []cellgrid.Diff
	for _, d := range diffs {
		if d.Got.Rune == ' ' && d.Want.Rune == ' ' {
			continue
		}
		live = append(live, d)
	}
	fmt.Printf("%s: strict differences %d, space-cell token or bold only %d, differences under the live rule %d\n", gridFile[strings.LastIndex(gridFile, "/")+1:], len(diffs), len(diffs)-len(live), len(live))
	fmt.Print(cellgrid.FormatDiffs(live))
}
