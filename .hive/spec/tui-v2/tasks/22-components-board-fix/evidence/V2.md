## Run

The script makes a small Go program in a scratch folder below `.hive/tmp/`. The program compares the `dev` grid with the task grid cell by cell with `cellgrid.Compare`. Then the script runs the exporter again into the scratch folder and compares each grid file. Last, it lists each grid file that changed between `dev` and the task branch.

```sh
echo "commit $(git rev-parse --short HEAD)"
T=$(mktemp -d -p .hive/tmp)
mkdir "$T/griddiff"
cat > "$T/griddiff/main.go" <<'GO'
package main

import (
	"fmt"
	"os"

	"github.com/Harrison-Blair/hive/internal/lib/cellgrid"
)

func read(p string) cellgrid.Grid {
	f, err := os.Open(p)
	if err != nil {
		panic(err)
	}
	defer f.Close()
	g, err := cellgrid.ReadGrid(f)
	if err != nil {
		panic(err)
	}
	return g
}

func main() {
	diffs, err := cellgrid.Compare(read(os.Args[1]), read(os.Args[2]), nil)
	if err != nil {
		panic(err)
	}
	fmt.Printf("%d cells differ\n", len(diffs))
	fmt.Print(cellgrid.FormatDiffs(diffs))
}
GO
git show dev:.hive/spec/tui-v2/design/Components.grid > "$T/dev.grid"
go run "./$T/griddiff" "$T/dev.grid" .hive/spec/tui-v2/design/Components.grid; echo "exit $?"
mkdir "$T/out"
go run ./internal/boardexport/cmd/boardexport -out "$T/out" .hive/spec/tui-v2/design; echo "export exit $?"
for f in "$T"/out/*.grid; do cmp "$f" ".hive/spec/tui-v2/design/$(basename "$f")" || echo "differs $(basename "$f")"; done; echo "export cmp done"
git diff --name-only dev HEAD -- '.hive/spec/tui-v2/design/*.grid'; echo "exit $?"
rm -rf "$T"
```
