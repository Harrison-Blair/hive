package tui

import (
 "testing"
 "time"
 tea "charm.land/bubbletea/v2"
)

func TestVerifierDemoAndNormalRenderers(t *testing.T) {
 f, err := loadDemoFixture(designDir+"/Main.fixture.json")
 if err != nil { t.Fatal(err) }
 home := t.TempDir()
 demo := newDemoModel(home, &demoState{fixture:f, size:demoSize{Width:120,Height:40}})
 defer demo.Stop()
 t.Chdir(home)
 normal := New(home, func() time.Time { return f.now })
 defer normal.Stop()
 // Give the normal model exactly the same loader output and current UI state.
 // The normal model keeps demo == nil and uses its normal View path.
 normal.useDemoLoaders(f)
 normal.applyDemoUI(f.UI)
 normal.Update(tea.WindowSizeMsg{Width:120,Height:40})
 for _, load := range []tea.Cmd{normal.startHeaderLoad(),normal.startCurrentScreenLoad()} { normal.Update(load()) }
 if demo.View().Content != normal.View().Content { t.Fatalf("demo:\n%q\nnormal:\n%q",demo.View().Content,normal.View().Content) }
}
