package main
import("fmt";"os";"strings";"html";"github.com/charmbracelet/x/ansi";"github.com/Harrison-Blair/hive/internal/lib/cellgrid")
func main(){ b,e:=os.ReadFile(os.Args[1]);must(e); f,e:=os.Open(os.Args[2]);must(e); want,e:=cellgrid.ReadGrid(f);must(e);lines:=strings.Split(string(b),"\n");if len(lines)<want.Height{panic("short capture")};for i:=0;i<want.Height;i++{lines[i]=ansi.Truncate(lines[i],want.Width,"")};got,e:=cellgrid.Convert(strings.Join(lines[:want.Height],"\n"),want.Width,want.Height);must(e);diff:=0;for i,c:=range got.Cells{w:=want.Cells[i];if c.Rune==' '&&w.Rune==' '{continue};if c!=w{fmt.Printf("row %d col %d want %+v got %+v\n",i/want.Width,i%want.Width,w,c);diff++}};fmt.Printf("%d cells differ\n",diff);colors:=[]string{"#e0ddd5","#3d3a33","#a39d90","#f5b942","#6ee69a","#7ccf6a","#6fb7ff","#e9cf9b","#9fd0ff"};var h strings.Builder;h.WriteString("<html><body style='margin:0;background:#161510'><pre style='margin:16px;font:16px/19px monospace'>");for i,c:=range got.Cells{weight:="normal";if c.Bold{weight="bold"};fmt.Fprintf(&h,"<span style='color:%s;font-weight:%s'>%s</span>",colors[int(c.Token)],weight,html.EscapeString(string(c.Rune)));if i%want.Width==want.Width-1{h.WriteByte('\n')}};h.WriteString("</pre></body></html>");must(os.WriteFile(os.Args[3],[]byte(h.String()),0600));if diff>0{os.Exit(1)}}
func must(e error){if e!=nil{panic(e)}}
