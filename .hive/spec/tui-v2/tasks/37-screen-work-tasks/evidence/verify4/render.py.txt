import re, html
from pathlib import Path
p=Path('.hive/spec/tui-v2/tasks/37-screen-work-tasks/evidence/verify4')
s=(p/'Work.ansi').read_text()
fg='#e6e1d5'; bold=False; out=[]; pos=0
for m in re.finditer(r'\x1b\[([0-9;]*)m',s):
    out.append(f'<span style="color:{fg};font-weight:{700 if bold else 400}">'+html.escape(s[pos:m.start()])+'</span>')
    codes=[int(x or 0) for x in m[1].split(';')]; i=0
    while i<len(codes):
        c=codes[i]
        if c==0: fg='#e6e1d5'; bold=False
        elif c==1: bold=True
        elif c==22: bold=False
        elif c==39: fg='#e6e1d5'
        elif c==38 and i+4<len(codes) and codes[i+1]==2:
            fg='#%02x%02x%02x'%tuple(codes[i+2:i+5]); i+=4
        i+=1
    pos=m.end()
out.append(html.escape(s[pos:]))
Path('.hive/tmp/verify4/work.html').write_text('<html><body style="margin:0;background:#171714"><pre style="margin:10px;font-family:monospace;font-size:14px;line-height:18px">'+''.join(out)+'</pre></body></html>')
plain=re.sub(r'\x1b\[[0-9;]*[A-Za-z]', '', s)
(p/'Work-plain.txt').write_text(plain)
for f in ['Work-deps','Work-brief','Work-verified']:
    plain=re.sub(r'\x1b\[[0-9;]*[A-Za-z]', '', (p/(f+'.ansi')).read_text())
    print(f, '\n'.join(x for x in plain.splitlines() if any(k in x for k in ['dependency graph','NEEDS','BLOCKS','Restart of','VERIFIED 3','Result: pass','▶ 2-'])))
