```sh
echo "commit $(git rev-parse --short HEAD)"
OUT=$(mktemp -d)
go run ./internal/boardexport/cmd/boardexport -out "$OUT" .hive/spec/tui-v2/design; echo "export exit $?"
ls "$OUT" | wc -l
for f in "$OUT"/*.grid; do cmp "$f" ".hive/spec/tui-v2/design/$(basename "$f")" && echo "same $(basename "$f")"; done
cat > "$OUT/rows.js" <<'JS'
const fs = require('fs'), vm = require('vm');
const page = fs.readFileSync(process.argv[2], 'utf8');
const src = page.match(/<script type="text\/x-dc" data-dc-script[^>]*>([\s\S]*?)<\/script>/)[1];
const ctx = { DCLogic: class {} }; vm.createContext(ctx);
const out = vm.runInContext(src + '\n;new Component().renderVals();', ctx);
for (const y of process.argv.slice(3)) console.log(y + '\t' + out.rows[+y].runs.map(r => r.text).join(''));
JS
node "$OUT/rows.js" .hive/spec/tui-v2/design/Main.dc.html 0 1 7 14 37 > "$OUT/rows.txt"
awk '/^t\|/{print (n++) "\t" substr($0, 3)}' .hive/spec/tui-v2/design/Main.grid | awk -F'\t' '$1==0||$1==1||$1==7||$1==14||$1==37' > "$OUT/grid.txt"
diff "$OUT/rows.txt" "$OUT/grid.txt" && echo "five rows match"
sed 's/^/rows() /' "$OUT/rows.txt"
sed 's/^/grid t| /' "$OUT/grid.txt"
rm -r "$OUT"
```
