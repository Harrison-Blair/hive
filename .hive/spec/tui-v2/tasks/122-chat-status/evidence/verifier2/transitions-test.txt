package tui

import (
 "strings"
 "testing"
 "time"
 tea "charm.land/bubbletea/v2"
 chatlib "github.com/Harrison-Blair/hive/internal/chat"
 "github.com/Harrison-Blair/hive/internal/lib/records"
 "github.com/charmbracelet/x/ansi"
)

func TestVerifierChatStatusTransitions(t *testing.T) {
 home := newTestHome(t)
 clock := fixedClock(t, time.Date(2026,10,20,12,0,59,0,time.UTC))
 writeRecord(t,home,[]string{"agents"},"coordinator-g1",records.Agent{
  Key:"coordinator-g1",Name:"coordinator",Role:"coordinator",Status:records.AgentLive,
  AgentStatus:"working",StatusChangedAt:"2026-10-20T12:00:00Z",
 })
 m:=newTestModel(t,home,clock)
 m,_=raise(t,m,tea.WindowSizeMsg{Width:120,Height:40})
 var err error
 m.header,err=readHeader(home)
 if err!=nil {t.Fatal(err)}
 m.chatData.entries=[]chatlib.Entry{
  {ID:"u",At:"2026-10-20T12:00:00Z",From:"user",Text:"draft",Delivery:"pending"},
  {ID:"a",At:"2026-10-20T12:00:30Z",From:"coordinator",Text:"answer"},
 }
 m.invalidateView()
 requireView(t,m,"message · ● working 0m · queued")
 clock.set(time.Date(2026,10,20,12,1,0,0,time.UTC))
 m,_=raise(t,m,tickMsg{})
 requireView(t,m,"message · ● working 1m · queued")
 m.chatData.entries[0].Delivery="sent"
 m.invalidateView()
 requireView(t,m,"message · ● working 1m")
 if strings.Contains(ansi.Strip(m.View().Content),"· queued") {t.Fatal("queued mark remains after delivery")}
 m.chatData.entries=append(m.chatData.entries,chatlib.Entry{
  ID:"new",At:"2026-10-20T12:00:45Z",From:"user",Text:"new draft",Delivery:"pending",
 })
 m.invalidateView()
 requireView(t,m,"message · ● working 1m · queued")
 m.chatData.entries[2].Delivery="failed"
 m.invalidateView()
 if strings.Contains(ansi.Strip(m.View().Content),"· queued") {t.Fatal("failed delivery shows queued")}
}
