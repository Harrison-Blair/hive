from live import *
from collections import defaultdict
folder=E/'multiline';results=[]
for case in ['short','tall','OtherNote']:
 for w,h in [(120,40),(80,24)]:
  paths=list(folder.glob(f'{case}-{w}-*.txt'))
  paths=[p for p in paths if 'editor' not in p.name]
  assert paths,(case,w)
  seen=set();first=False
  for p in sorted(paths):
   lines=p.read_text().splitlines();assert len(lines)==h,(p,len(lines))
   assert all(len(line)<=w for line in lines),(p,'line width')
   start=next(i for i,l in enumerate(lines) if l.startswith('┏'))
   end=next(i for i,l in enumerate(lines) if i>start and l.startswith('┗'))
   assert end-start+1<=h//2,(p,'dock overflow')
   assert ' [review]' in lines[-1],p
   for line in lines[start+1:end]:
    assert line.startswith('┃') and line.endswith('┃'),(p,line)
    assert '…' not in line,(p,'cut row')
    seen.update(line.strip('┃ ›').split())
   if '› Label → ROW01' in '\n'.join(lines):first=True
  if case=='tall':
   assert {f'ROW{i:02}' for i in range(1,31)}|{'FINAL-LABEL-SENTINEL'}<=seen,(w,seen)
   assert first,(w,'first line not highlighted')
  if case=='short':assert {'First','line','Second'}<=seen
  if case=='OtherNote':
   for p in paths:
    s=p.read_text()
    for text in ['Second line, Click to focus, other one','other two · note: note one','note two','tail note one','tail note two']:
     assert text in s,(p,text)
  results.append(f'{case} {w}x{h}: {len(paths)} captures, borders intact, dock <= {h//2} rows, frame fits, no cut. '+('ROW01 to ROW30 and FINAL-LABEL-SENTINEL reached with arrows.' if case=='tall' else 'All expected text visible.'))
(E/'multiline-check.txt').write_text('\n'.join(results)+'\n');print('\n'.join(results))
