#!/bin/bash
set -u
commands=("gofmt -l \$(git ls-files -co --exclude-standard '*.go')" "go mod tidy -diff" "go vet ./..." "golangci-lint run ./..." "go test -race ./...")
failed=0
for cmd in "${commands[@]}"; do
 printf '\n$ %s\n' "$cmd"
 bash -c "$cmd"
 rc=$?
 printf 'exit=%s\n' "$rc"
 if [ "$rc" != 0 ]; then failed=1; fi
done
exit "$failed"
