package tui

import (
 "fmt"
 "os"
 "strings"
 "testing"
 "github.com/Harrison-Blair/hive/internal/ask"
 chatlib "github.com/Harrison-Blair/hive/internal/chat"
 "github.com/Harrison-Blair/hive/internal/lib/records"
 "github.com/charmbracelet/x/ansi"
)

// A stopped review must retain a plain-text state after ordinary later chat
// messages move the original ask event above the feed viewport.
func TestVerifierStoppedReviewWhenNeitherCaptionFits(t *testing.T) {
 const asker = "asker-scratch-with-long-name-abc"
 const parent = "parent-scratch"
 for _, size := range [][2]int{{80,24},{120,40}} {
  t.Run(fmt.Sprint(size[0]),func(t *testing.T) {
   a:=oneQuestionAsk("a1",asker);a.Parent=parent;a.ParentKey=parent
   m,calls:=askModel(t,size[0],size[1],a)
   agents:=[]records.Agent{{Key:asker,Name:asker,Status:records.AgentStopped},{Key:parent,Name:parent,Status:records.AgentLive}}
   var entries []chatlib.Entry
   for i:=1;i<=30;i++ {entries=append(entries,chatlib.Entry{ID:fmt.Sprintf("c%d",i),At:fmt.Sprintf("2026-10-20T11:40:%02dZ",i),From:"coordinator",Text:fmt.Sprintf("Later chat message %d",i)})}
   m.chatData=buildChatScreenData(entries,nil,[]ask.Record{a},agents,m.now());m.rebuildChat();m.invalidateView()
   m=run(t,m,key('1',"1"));requireNoCalls(t,calls,"review")
   view:=ansi.Strip(m.View().Content)
   dir:=os.Getenv("VERIFIER_CAPTURE");if dir!="" {if err:=os.WriteFile(fmt.Sprintf("%s/no-caption-%d.txt",dir,size[0]),[]byte(view),0600);err!=nil {t.Fatal(err)}}
   requireFits(t,m,size[0],size[1])
   requireView(t,m,"? "+asker+" asks · confirm","Send to "+parent,"enter sends to "+parent,"[review]")
   var card []string;inside:=false
   for _,line:=range strings.Split(view,"\n") {if strings.HasPrefix(line,"┏") {inside=true};if inside {card=append(card,line)};if strings.HasPrefix(line,"┗") {inside=false}}
   if !strings.Contains(strings.Join(card,"\n"),"asker stopped") {t.Fatalf("review card does not identify the asker as stopped:\n%s",strings.Join(card,"\n"))}
  })
 }
}
