def crop(text,w,h):
 out=[]
 for line in text.splitlines()[:h]:
  parts=re.findall(r'\x1b\[[0-9;]*m|[^\x1b]',line);col=0;s=''
  for p in parts:
   if p.startswith('\x1b'):s+=p
   elif col<w:s+=p;col+=1
   else:break
  out.append(s+'\x1b[0m')
 return '\n'.join(out)+'\n'
