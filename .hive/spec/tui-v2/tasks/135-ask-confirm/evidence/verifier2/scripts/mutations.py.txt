from pathlib import Path
import subprocess
root=Path.cwd(); scratch=root/'.hive/tmp/verifier135b'; out=root/'.hive/spec/tui-v2/tasks/135-ask-confirm/evidence/verifier2'
# The source copy already contains the checked task commit.
p=scratch/'current/internal/tui/ask.go'; original=p.read_text()
mutations=[
('empty-other', [('card.editing == "other" && strings.TrimSpace(*text) == ""','false && card.editing == "other" && strings.TrimSpace(*text) == ""'), ('card.picks[q][other] && strings.TrimSpace(card.others[q]) == ""','false && card.picks[q][other] && strings.TrimSpace(card.others[q]) == ""')], 'TestAskReviewKeepsTheEmptyOtherCheck'),
('blocked-caller', [('if m.chatCallerBlocked {','if false {')], 'TestAskReviewBlockedCallerSendsNothing'),
('review-digit', [('case "up":\n\t\tswitch {','case "2":\n\t\treturn m.askPick(record, card, 1), true\n\tcase "up":\n\t\tswitch {')], 'TestAskReviewIgnoresOtherInput'),
('switch-draft', [('m.askFocusID = open[index].ID','clear(m.askCards)\n\tm.askFocusID = open[index].ID')], 'TestAskReviewSurvivesACardSwitch'),
('refresh-draft', [('if ok && len(card.cursor) == len(record.Questions) {','if false && ok && len(card.cursor) == len(record.Questions) {')], 'TestAskReviewSurvivesARefresh'),
('edit-return', [('q < len(record.Questions)-1 && !card.fromReview','q < len(record.Questions)-1')], 'TestAskReviewEditsAQuestionAndSendsTheEditedDraft'),
('cut-row', [('for row, line := range wrapAskWords(words, room-2) {','for row, line := range wrapAskWords(words, room-2)[:1] {')], 'TestAskReviewWrapsAndScrollsLongContent'),
('frame-keys', [('if card.review {\n\t\treturn m.updateAskReviewKey(press, record, card)\n\t}', 'if card.review {\n\t\tswitch key { case "i", "ctrl+o", "pgup", "pgdown", "?": return nil, false }\n\t\treturn m.updateAskReviewKey(press, record, card)\n\t}')], 'TestAskReviewKeepsOnlyTheReviewKeys'),
('tall-row', [('case card.reviewScroll < card.reviewMax:', 'case false:')], 'TestAskReviewReachesEachPartOfALongRow'),
]
with (out/'mutations.txt').open('w') as log:
 for name,replacements,test in mutations:
  source=original
  for old,new in replacements:
   assert source.count(old)==1,(name,old,source.count(old))
   source=source.replace(old,new)
  p.write_text(source)
  r=subprocess.run(['go','test','./internal/tui','-count=1','-run','^'+test+'$'],cwd=scratch/'current',text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
  log.write(f'\nMUTATION {name}: {test}, exit={r.returncode}\n{r.stdout}');log.flush()
  p.write_text(original)
  assert r.returncode!=0 and 'build failed' not in r.stdout,(name,r.stdout)
  print(name,'detected',flush=True)
assert p.read_text()==original
