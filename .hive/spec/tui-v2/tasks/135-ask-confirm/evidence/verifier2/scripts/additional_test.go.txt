func TestVerifierMultilineReviewFieldsAndReplies(t *testing.T) {
 for _, size := range [][2]int{{120,40},{80,24}} {
  t.Run(fmt.Sprint(size[0]),func(t *testing.T) {
   data:=[]byte(`{"questions":[{"question":"Which label?","header":"Head\nTail","multiSelect":true,"options":[{"label":"first\n\nthird","description":"multiline"},{"label":"No","description":"no"}]}]}`)
   req,err:=ask.Parse(data);if err!=nil {t.Fatal(err)}
   record:=oneQuestionAsk("a1","manager-hive");record.Questions=req.Questions
   m,calls:=askModel(t,size[0],size[1],record)
   m=run(t,m,key('1',"1"));m=run(t,m,key('3',"3"))
   m=run(t,m,tea.KeyPressMsg{Code:'o',Text:"other one\nother two"})
   m=run(t,m,ctrlBracket());m=run(t,m,tea.KeyPressMsg{Code:tea.KeyTab})
   m=run(t,m,tea.KeyPressMsg{Code:'n',Text:"note one\nnote two"});m=run(t,m,enterKey)
   requireNoCalls(t,calls,"review")
   seen:=""
   for i:=0;i<5;i++ {
    body:=requireReviewDock(t,m,size[1]/2)
    for _,line:=range body {if strings.Contains(line,"…"){t.Fatalf("truncated row: %q",line)}}
    seen+=strings.Join(body,"\n")+"\n"
    m=run(t,m,tea.KeyPressMsg{Code:tea.KeyUp})
   }
   for _,s:=range []string{"Head ","┃   Tail → first ","┃   third, other one ","┃   other two · note: note one ","┃   note two "} {
    if !strings.Contains(seen,s){t.Fatalf("missing %q in %s",s,seen)}
   }
   for i:=0;i<5;i++ {m=run(t,m,tea.KeyPressMsg{Code:tea.KeyDown})}
   run(t,m,enterKey)
   requireCalls(t,calls,[]askCall{{"answer","a1",[]ask.Reply{{Picks:[]int{1,3},Other:"other one\nother two",Note:"note one\nnote two"}}}})
  })
 }
}

// A new stopped-asker review must retain the complete required state title.
func TestVerifierCompactStoppedReviewTitle(t *testing.T) {
 record:=oneQuestionAsk("a1","asker-scratch");record.Parent="parent-scratch";record.ParentKey="parent-scratch"
 m,_:=askModel(t,80,24,record)
 agents:=[]records.Agent{{Key:"asker-scratch",Name:"asker-scratch",Status:records.AgentStopped},{Key:"parent-scratch",Name:"parent-scratch",Status:records.AgentLive}}
 m.chatData=buildChatScreenData(m.chatData.entries,nil,[]ask.Record{record},agents,m.now());m.rebuildChat()
 m=run(t,m,key('1',"1"))
 view:=ansi.Strip(m.View().Content)
 for _,line:=range strings.Split(view,"\n") {if strings.HasPrefix(line,"┏"){t.Log(line)}}
 requireView(t,m,"? asker-scratch asks · confirm","asker stopped · answer goes to parent-scratch","[review]")
}

// Render the same existing editor states on dev and the task without sending.
func TestVerifierExistingMultilineEditorStates(t *testing.T) {
 out:=os.Getenv("VERIFIER_CAPTURE");if out=="" {t.Fatal("VERIFIER_CAPTURE is required")}
 for _,size:=range [][2]int{{120,40},{80,24}} {
  for _,kind:=range []string{"label","header","stopped"} {
   t.Run(fmt.Sprintf("%s-%d",kind,size[0]),func(t *testing.T) {
    record:=oneQuestionAsk("a1","asker-scratch")
    if kind=="label" {record.Questions[0].Options[0].Label="First line\nSecond line"}
    if kind=="header" {record.Questions[0].Header="Head\nTail"}
    request:=ask.Request{Questions:record.Questions};data,err:=json.Marshal(request);if err!=nil {t.Fatal(err)}
    if _,err:=ask.Parse(data);err!=nil {t.Fatal(err)}
    m,_:=askModel(t,size[0],size[1],record)
    if kind=="stopped" {
     record.Parent="parent-scratch";record.ParentKey="parent-scratch"
     agents:=[]records.Agent{{Key:"asker-scratch",Name:"asker-scratch",Status:records.AgentStopped},{Key:"parent-scratch",Name:"parent-scratch",Status:records.AgentLive}}
     m.chatData=buildChatScreenData(m.chatData.entries,nil,[]ask.Record{record},agents,m.now());m.rebuildChat()
    }
    view:=m.View().Content
    if err:=os.WriteFile(fmt.Sprintf("%s/%s-%d.ansi",out,kind,size[0]),[]byte(view),0600);err!=nil {t.Fatal(err)}
    if err:=os.WriteFile(fmt.Sprintf("%s/%s-%d.txt",out,kind,size[0]),[]byte(ansi.Strip(view)),0600);err!=nil {t.Fatal(err)}
   })
  }
 }
}
