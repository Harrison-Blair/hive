def ansihtml(text):
 color='#e6e1d6';bold=False;out=[]
 for part in re.split(r'(\x1b\[[0-9;]*m)',text):
  if part.startswith('\x1b'):
   codes=[int(x or 0) for x in part[2:-1].split(';')];i=0
   while i<len(codes):
    c=codes[i]
    if c==0:color='#e6e1d6';bold=False
    elif c==1:bold=True
    elif c==22:bold=False
    elif c==39:color='#e6e1d6'
    elif c==38 and i+4<len(codes) and codes[i+1]==2:color='#%02x%02x%02x'%tuple(codes[i+2:i+5]);i+=4
    i+=1
  elif part:out.append(f'<span style="color:{color};font-weight:{700 if bold else 400}">{html.escape(part)}</span>')
 return ''.join(out)
