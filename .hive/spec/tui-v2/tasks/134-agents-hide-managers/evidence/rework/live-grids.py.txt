import json,subprocess,sys,time,os
W=os.getcwd(); T=W+'/.hive/tmp/rework134'; OUT=sys.argv[1]; os.makedirs(OUT,exist_ok=True)
D='.hive/spec/tui-v2/design'; N='.hive/spec/tui-v2/tasks/134-agents-hide-managers/evidence/new-grids'
cases=[(b,f'{D}/{b}.fixture.json',f'{N}/{b}.grid') for b in 'AgentsHiddenParent LiveOverview LivePod CompactLiveOverview CompactLivePod TreeNested TreeWide CompactTreeNested CompactTreeWide'.split()]
cases+=[(b,f'{N}/fixtures/{b}.fixture.json',f'{N}/{b}.grid') for b in 'HideManagerOverview CompactHideManagerOverview HideManagerPod CompactHideManagerPod HideAllPods CompactHideAllPods'.split()]
cases+=[(b,f'{D}/{b}.fixture.json',f'{D}/{b}.grid') for b in 'Main AgentsMany CompactAgents AgentsHome AgentsCoordinatorChild'.split()]
def h(*a):
    r=subprocess.run(['herdr',*a],capture_output=True,text=True)
    if r.returncode: raise SystemExit(f'herdr {a}: {r.stderr}')
    return r.stdout
res=json.loads(h('tab','create','--workspace','wWT','--cwd',W,'--label','rework134-grids'))['result']
pane=res['root_pane']['pane_id']; tabid=res['tab']['tab_id']
def read(fmt):
    prev=None
    for _ in range(20):
        time.sleep(0.5)
        cur=h('pane','read',pane,'--source','visible','--format',fmt)
        if cur==prev: return cur
        prev=cur
    return cur
log=[]; bad=0
try:
    for name,fx,grid in cases:
        t=json.load(open(fx))['terminal']; size=f"{t['width']}x{t['height']}"
        h('pane','run',pane,f'clear; HIVE_HOME={T}/home HIVE_NO_STARTUP=1 TERM=vt100 COLORTERM=truecolor {T}/hive --demo {W}/{fx} --size {size}')
        time.sleep(1.5)
        a=read('ansi'); open(f'{OUT}/{name}.ansi','w').write(a)
        open(f'{OUT}/{name}.txt','w').write(read('text'))
        r=subprocess.run([f'{T}/cmpgrid',f'{OUT}/{name}.ansi',grid,str(t['width']),str(t['height'])],capture_output=True,text=True)
        log.append(f'{name} {size} vs {grid}: {r.stdout.strip().splitlines()[0] if r.stdout else r.stderr}')
        if r.returncode: bad+=1; open(f'{OUT}/{name}.diff.txt','w').write(r.stdout)
        h('pane','send-keys',pane,'ctrl+c'); time.sleep(0.3); h('pane','send-keys',pane,'ctrl+c'); time.sleep(1)
        assert 'project agents' not in read('text'), 'tui still running'
finally:
    h('tab','close',tabid)
log.append(f'{len(cases)} comparisons, {bad} with differences')
open(f'{OUT}/grid-compare.log','w').write('\n'.join(log)+'\n'); print('\n'.join(log))
