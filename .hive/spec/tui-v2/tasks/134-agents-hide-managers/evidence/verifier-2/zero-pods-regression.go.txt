package tui

import (
 "fmt"
 "testing"
 tea "charm.land/bubbletea/v2"
)

// Check the zero-pod path without stubbing the model or its record loader.
func TestVerifierZeroPodOverviewKeys(t *testing.T) {
 for _, size := range hideSizes {
  t.Run(fmt.Sprintf("%dx%d", size.Width, size.Height), func(t *testing.T) {
   home, clock := coordinatorOnlyHome(t)
   m := openHide(t, home, clock, size)
   for _, k := range []tea.KeyPressMsg{key(tea.KeyLeft,""),key(tea.KeyRight,""),key(tea.KeyDown,""),key(tea.KeyEnter,"")} {
    m,_ = raise(t,m,k)
    requireView(t,m,"no project agents","● coordinator")
    rejectView(t,m,"[details]","esc closes")
    if m.graph.podView {t.Fatal("empty enter or arrow opened a phantom pod")}
   }
   for _, up := range []tea.KeyPressMsg{key(tea.KeyUp,""),key('k',"k")} {
    m,_=raise(t,m,up)
    requireCoordinatorHeavy(t,m,size,true)
    for _, k := range []tea.KeyPressMsg{key(tea.KeyLeft,""),key(tea.KeyRight,"")} {
     m,_=raise(t,m,k);requireCoordinatorHeavy(t,m,size,true)
    }
    m,_=raise(t,m,key(tea.KeyDown,""))
    requireCoordinatorHeavy(t,m,size,false)
    m,_=raise(t,m,key(tea.KeyEnter,""))
    rejectView(t,m,"[details]","esc closes")
    if m.graph.podView {t.Fatal("enter with no selected coordinator opened a phantom pod")}
   }
   // A previous pod view must not block keys after its last pod hides.
   m.graph.podView=true
   m,_=raise(t,m,key('k',"k"))
   requireCoordinatorHeavy(t,m,size,true)
   m,_=raise(t,m,key(tea.KeyEnter,""))
   requireView(t,m,"coordinator · esc closes","[details]")
  })
 }
}
