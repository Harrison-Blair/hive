import pathlib,subprocess,concurrent.futures,json,shutil
W=pathlib.Path.cwd();T=W/'.hive/tmp/verify134-second';E=W/'.hive/spec/tui-v2/tasks/134-agents-hide-managers/evidence/verifier-2';L=E/'live';H=T/'capture-html';H.mkdir(exist_ok=True)
initial='AgentsHiddenParent LiveOverview LivePod CompactLiveOverview CompactLivePod TreeNested TreeWide CompactTreeNested CompactTreeWide HideManagerOverview CompactHideManagerOverview HideManagerPod CompactHideManagerPod HideAllPods CompactHideAllPods Main AgentsMany CompactAgents AgentsHome AgentsCoordinatorChild'.split()
keys=[]
for size in ('120x40','80x24'):
 keys += [f'zero-pods-{size}-{s}' for s in ('0-start','5-up','8-enter','9-esc','12-k','13-enter-k','16-h-hides','17-enter-after-cycle')]
 keys += [f'selected-before-{size}-4-enter',f'no-registry-{size}-7-enter',f'nothing-visible-{size}-7-enter',f'home-only-{size}-1-enter']
names=initial+keys
for size in ('120x40','80x24'):
 names += [f'general-child-{size}-1-right', f'general-child-{size}-2-enter', f'pod-selection-{size}-4-h-restores']
 names += [f'manager-{state}-{size}' for state in ('stopped','exited','stale')]

def job(name):
 if (L/(name+'.png')).exists():return name+': PNG already captured, '+str((L/(name+'.png')).stat().st_size)+' bytes'
 a=L/(name+'.ansi');meta=L/(name+'.metadata.json')
 if not a.exists():return name+': not present yet'
 size=json.loads(meta.read_text())['size'];w,h=size.split('x');copy=H/a.name;shutil.copyfile(a,copy)
 r=subprocess.run([str(T/'gridtool'),str(copy),'-',w,h],capture_output=True,text=True);assert r.returncode==0,(name,r.stderr)
 profile=T/('screen-profile-'+name);profile.mkdir(exist_ok=True)
 cmd=['firefox','--headless','--no-remote','--profile',str(profile),'--window-size=1220,780','--screenshot',str(L/(name+'.png')),'file://'+str(copy)+'.html']
 r=subprocess.run(cmd,capture_output=True,text=True,timeout=90);assert r.returncode==0,(name,r.stderr)
 return name+': PNG captured, '+str((L/(name+'.png')).stat().st_size)+' bytes'
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as ex:out=list(ex.map(job,names))
(E/'screenshots.log').write_text('\n'.join(out)+'\n');print('\n'.join(out))
