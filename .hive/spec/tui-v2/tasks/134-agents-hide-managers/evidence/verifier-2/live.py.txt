def heavy(txt):
 for line in txt.splitlines():
  i=line.find('● coordinator')
  if i>=2:return line[i-2]=='┃'

def check(txt,require=(),reject=(),selected=None):
 for s in require:assert s in txt,('missing',s,txt)
 for s in reject:assert s not in txt,('unexpected',s,txt)
 if selected is not None:assert heavy(txt)==selected,('heavy',selected,txt)
 assert 'panic:' not in txt and 'fatal error' not in txt
