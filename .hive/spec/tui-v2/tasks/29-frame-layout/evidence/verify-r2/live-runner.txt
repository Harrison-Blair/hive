import subprocess,time,json
from pathlib import Path
root=Path.cwd(); ev=root/'.hive/spec/tui-v2/tasks/29-frame-layout/evidence/verify-r2'; pane='w8T:p2'
def call(*args): return subprocess.check_output(['herdr',*args],text=True)
def capture(name):
 time.sleep(.7)
 a=call('pane','read',pane,'--source','visible','--format','ansi'); time.sleep(.5)
 b=call('pane','read',pane,'--source','visible','--format','ansi'); assert a==b, name+' unstable'
 meta=json.loads(call('pane','get',pane)); assert meta['result']['pane']['scroll']['offset_from_bottom']==0
 assert not meta['result'].get('truncated',False)
 (ev/(name+'.ansi')).write_text(b); (ev/(name+'-pane.json')).write_text(json.dumps(meta))
def keys(name,*ks):
 for k in ks: call('pane','send-keys',pane,k); time.sleep(.15)
 capture(name)
def quit(name):
 keys(name+'-quit-notice','ctrl+c'); keys(name+'-shell','ctrl+c')
def start(board,size):
 # The previous program has exited; verify shell before sending a command.
 info=json.loads(call('pane','process-info','--pane',pane)); assert all(p['name']=='bash' for p in info['result']['process_info']['foreground_processes'])
 call('pane','run',pane,f'{root}/.hive/tmp/verify29r2/hive --demo {root}/.hive/spec/tui-v2/design/{board}.fixture.json --size {size}')
 capture(board+'-live')
capture('Main-live')
keys('Main-tab','tab'); keys('Main-help','?'); keys('Main-help-scroll','down'); keys('Main-help-closed','esc')
keys('Main-work','3'); keys('Main-work-details','tab'); keys('Main-work-list','tab'); quit('Main')
start('Help','120x40'); keys('Help-key-2','2'); keys('Help-closed-question','?'); keys('Help-reopened','?'); keys('Help-closed','esc'); quit('Help')
start('CompactAgents','80x24'); keys('CompactAgents-tab','tab'); keys('CompactAgents-popup','enter'); keys('CompactAgents-popup-3','3'); keys('CompactAgents-closed','esc'); keys('CompactAgents-help','?'); keys('CompactAgents-help-closed','esc'); quit('CompactAgents')
start('Compact','80x24'); keys('Compact-key-1','1'); keys('Compact-blocked-keys','6','[',']','tab'); keys('Compact-list','esc'); keys('Compact-reopened','enter'); keys('Compact-list-again','esc'); keys('Compact-list-1','1'); quit('Compact')
start('TooSmall','60x16'); quit('TooSmall')
print('Live capture and keys complete')
