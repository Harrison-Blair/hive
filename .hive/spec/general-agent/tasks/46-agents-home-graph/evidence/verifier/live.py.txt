# Render visible terminal SGR to HTML. Preserve each cell and foreground/bold.
def terminal_html(raw,width,height):
 out=[]; color='#ddd9cf'; bold=False
 for line in raw.splitlines()[:height]:
  cells=0
  for bit in re.split(r'(\x1b\[[0-9;]*m)',line):
   if bit.startswith('\x1b['):
    codes=[int(c or 0) for c in bit[2:-1].split(';')]; i=0
    while i<len(codes):
     c=codes[i]
     if c==0: color='#ddd9cf'; bold=False
     elif c==1: bold=True
     elif c==22: bold=False
     elif c==39: color='#ddd9cf'
     elif c==38 and codes[i+1]==2: color='#%02x%02x%02x'%tuple(codes[i+2:i+5]); i+=4
     i+=1
   else:
    text=bit[:max(0,width-cells)]; cells+=len(text); out.append('<span style="color:'+color+';font-weight:'+('700' if bold else '400')+'">'+html.escape(text)+'</span>')
  out.append('\n')
 return '<html><body style="margin:0;background:#171612"><pre style="margin:8px;font-family:monospace;font-size:13px;line-height:18px">'+''.join(out)+'</pre></body></html>'
