from pathlib import Path
import re, subprocess
base = subprocess.check_output(['git', 'show', '72a33b5b:reference/dogfood/friction.md'], text=True)
def source_entries(text):
    found = {}
    for section in re.finditer(r'(?ms)^## (?:Open|Resolved)\n(.*?)(?=^## |\Z)', text):
        for item in re.finditer(r'(?ms)^### (.*?)\n(.*?)(?=^### |\Z)', section.group(1)):
            title, body = item.group(1), item.group(2)
            summary = body.split('**Summary:**', 1)[1].lstrip(' \n').split('\n\n**Reproduction steps:**\n', 1)[0].rstrip()
            steps = body.split('\n\n**Reproduction steps:**\n', 1)[1].rstrip()
            found[title] = (summary, steps)
    return found
original = source_entries(base)
files = list(Path('reference/dogfood/friction/entries').rglob('*.md'))
current = {}
for path in files:
    text = path.read_text()
    title = re.search(r'^title: "(.*)"$', text, re.M).group(1)
    summary = text.split('**Summary:**\n', 1)[1].split('\n\n**Reproduction steps:**\n', 1)[0].rstrip()
    steps = text.split('\n\n**Reproduction steps:**\n', 1)[1].rstrip()
    current[title] = (summary, steps)
merged_title = 'An agent spawn can skip the first prompt'

assert (summary_prompt, steps_prompt) == original[merged_title]

    assert current[title] == expected, title
print(f'PASS: all {len(original)} source entries retain their summary text and reproduction steps.')
