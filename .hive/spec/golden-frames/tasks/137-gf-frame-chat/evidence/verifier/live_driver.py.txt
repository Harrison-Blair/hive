import fcntl, json, os, pathlib, pty, re, select, shlex, socket, struct, subprocess, termios, threading, time
ROOT = pathlib.Path.cwd()
OUT = ROOT / '.hive/spec/golden-frames/tasks/137-gf-frame-chat/evidence/verifier/live'
OUT.mkdir(parents=True, exist_ok=True)
SESSION = 't137-check-verifier'
SOCK = str(pathlib.Path.home() / '.config/herdr/sessions' / SESSION / 'herdr.sock')
seq = 0

def request_once(method, params):
    global seq
    seq += 1
    rid = 't137-verify-' + str(seq)
    deadline = time.monotonic() + 10
    with socket.socket(socket.AF_UNIX) as s:
        s.settimeout(max(deadline - time.monotonic(), 0.001))
        s.connect(SOCK)
        s.sendall((json.dumps({'id': rid, 'method': method, 'params': params}) + '\n').encode())
        buf = b''
        while not buf.endswith(b'\n'):
            s.settimeout(max(deadline - time.monotonic(), 0.001))
            data = s.recv(65536)
            if not data: raise RuntimeError('EOF')
            buf += data
            if len(buf) > 4 * 1024 * 1024: raise RuntimeError('large response')
        obj = json.loads(buf)
        if obj.get('id') != rid: raise RuntimeError('wrong id')
        if 'error' in obj: raise RuntimeError(obj['error'])
        return obj['result']

def cli(*args):
    env = dict(os.environ, HERDR_SOCKET_PATH=SOCK)
    return subprocess.run(['herdr', *args], env=env, check=True, text=True, capture_output=True, timeout=15).stdout

def read():
    data = request_once('pane.read', {'pane_id': PANE, 'source': 'visible', 'format': 'ansi'})['read']
    pane = request_once('pane.get', {'pane_id': PANE})['pane']
    assert not data['truncated'] and pane['scroll']['offset_from_bottom'] == 0
    return data['text']

def stable():
    prev = read()
    for _ in range(30):
        time.sleep(.5)
        cur = read()
        if cur == prev: return cur
        prev = cur
    raise RuntimeError('screen did not settle')

def keys(*args):
    cli('pane', 'send-keys', PANE, *args)
    time.sleep(.25)

def save(name, text):
    (OUT / (name + '.ansi')).write_text(text)
    plain = re.sub(r'\x1b\[[0-?]*[ -/]*[@-~]', '', text)
    (OUT / (name + '.txt')).write_text(plain)
    return plain

def start(board):
    command = 'clear; TERM=vt100 COLORTERM=truecolor HIVE_HOME=' + shlex.quote(str(ROOT / '.hive/tmp/t137-check-verifier/home')) + ' ' + shlex.quote(str(ROOT / '.hive/tmp/t137-check-verifier/hive')) + ' --demo ' + shlex.quote(str(ROOT / '.hive/spec/golden-frames/design' / (board + '.fixture.json'))) + ' --size 120x40'
    cli('pane', 'send-text', PANE, command)
    keys('enter')
    time.sleep(1)
    view = stable()
    if 'not a hive home' in view or 'terminal too small' in view or '╭' not in view:
        raise RuntimeError('demo did not start: ' + view)
    return view

def quit():
    keys('ctrl+c')
    keys('ctrl+c')
    time.sleep(.6)

master, slave = pty.openpty()
fcntl.ioctl(slave, termios.TIOCSWINSZ, struct.pack('HHHH', 50, 170, 0, 0))
env = {k: v for k, v in os.environ.items() if not k.startswith('HERDR_')}
env['TERM'] = 'xterm-256color'
client = subprocess.Popen(['herdr', '--session', SESSION], stdin=slave, stdout=slave, stderr=slave, env=env, start_new_session=True)
os.close(slave)
stop = threading.Event()
def drain():
    while not stop.is_set():
        if select.select([master], [], [], .2)[0]:
            try: os.read(master, 65536)
            except OSError: return
thread = threading.Thread(target=drain, daemon=True)
thread.start()
try:
    for _ in range(60):
        if os.path.exists(SOCK): break
        time.sleep(.25)
    ping = request_once('ping', {})
    assert ping['type'] == 'pong' and ping['protocol'] == 22
    panes = json.loads(cli('pane', 'list'))['result']['panes']
    assert len(panes) == 1, panes
    PANE = panes[0]['pane_id']
    cli('pane', 'rename', PANE, 't137-check-verifier-live')
    layout = cli('pane', 'layout', '--pane', PANE)
    (OUT / 'session.txt').write_text(json.dumps(ping) + '\n' + layout)
    boards = sorted(p.stem.replace('.fixture', '') for p in (ROOT / '.hive/spec/golden-frames/design').glob('*.fixture.json'))
    for board in boards:
        save(board + '-120x40', start(board))
        quit()
        print('captured', board, flush=True)
    save('keys-start', start('GfChat'))
    for key, name in [('ctrl+g','agents'),('ctrl+g','agents-back'),('ctrl+t','work'),('esc','work-esc'),('ctrl+f','friction'),('esc','friction-esc'),('ctrl+y','health'),('esc','health-esc'),('ctrl+l','logs'),('ctrl+l','logs-back'),('esc','chat-esc'),('ctrl+]','box-noop'),('ctrl+o','events'),('ctrl+o','events-folded'),('ctrl+c','quit-notice')]:
        keys(key)
        save('keys-' + name, stable())
    keys('ctrl+c')
    time.sleep(.8)
    save('keys-quit', stable())
    save('ask-start', start('GfChatAsk'))
    for key, name in [('ctrl+r','input'),('ctrl+r','card'),('esc','esc-input')]:
        keys(key)
        save('ask-' + name, stable())
    quit()
    save('empty-start', start('GfChatEmpty'))
    keys('?')
    save('empty-help', stable())
    keys('esc')
    save('empty-help-esc', stable())
    quit()
finally:
    subprocess.run(['herdr','session','stop',SESSION], check=True, timeout=15)
    try:
        client.wait(timeout=3)
    except subprocess.TimeoutExpired:
        client.terminate()
        client.wait(timeout=10)
    stop.set()
    os.close(master)
    subprocess.run(['herdr','session','delete',SESSION], check=True, timeout=15)
    print('closed scratch session', SESSION, flush=True)
