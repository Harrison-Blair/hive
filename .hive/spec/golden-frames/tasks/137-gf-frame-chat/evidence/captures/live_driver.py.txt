import json, socket, subprocess, sys, time, os
SOCK = os.environ["T137_SOCK"]
PANE = "w1:p1"
WT = "/home/penguin/source/hive/projects/hive/worktrees/137-gf-frame-chat"
SP = "SCRATCH"
OUT = sys.argv[1]
os.makedirs(OUT, exist_ok=True)
n = [0]
def call(method, params):
    n[0] += 1
    s = socket.socket(socket.AF_UNIX); s.connect(SOCK)
    s.sendall((json.dumps({"id": f"t137-{n[0]}", "method": method, "params": params}) + "\n").encode())
    buf = b""
    while not buf.endswith(b"\n"):
        c = s.recv(1 << 20)
        if not c: break
        buf += c
    s.close()
    r = json.loads(buf)
    if "error" in r: raise RuntimeError(r["error"])
    return r["result"]
def read():
    r = call("pane.read", {"pane_id": PANE, "source": "visible", "format": "ansi"})["read"]
    if r["truncated"]: raise RuntimeError("truncated read")
    p = call("pane.get", {"pane_id": PANE})["pane"]
    if p["scroll"]["offset_from_bottom"] != 0: raise RuntimeError("scroll offset")
    return r["text"]
def stable():
    prev = read()
    while True:
        time.sleep(0.5)
        cur = read()
        if cur == prev: return cur
        prev = cur
def keys(*k): subprocess.run(["env", "HERDR_SOCKET_PATH=" + SOCK, "herdr", "pane", "send-keys", PANE, *k], check=True, capture_output=True)
def text(t): subprocess.run(["env", "HERDR_SOCKET_PATH=" + SOCK, "herdr", "pane", "send-text", PANE, t], check=True, capture_output=True)
def start(board):
    text(f"clear; TERM=vt100 COLORTERM=truecolor HIVE_HOME={WT} {SP}/hive --demo {WT}/.hive/spec/golden-frames/design/{board}.fixture.json --size 120x40")
    keys("enter"); time.sleep(1.5)
    return stable()
def quit():
    keys("ctrl+c"); time.sleep(0.3); keys("ctrl+c"); time.sleep(1.0)
def save(name, data):
    open(os.path.join(OUT, name + ".ansi"), "w").write(data)
mode = sys.argv[2]
if mode == "boards":
    for b in sys.argv[3:]:
        save(b + "-120x40", start(b)); quit(); print("captured", b)
else:
    # mode keys: board, then key steps as name=key[,key]
    b = sys.argv[3]
    save(b + "-start", start(b))
    for step in sys.argv[4:]:
        name, ks = step.split("=", 1)
        for k in ks.split(","):
            keys(k); time.sleep(0.25)
        time.sleep(0.5)
        save(f"{b}-{name}", stable()); print("captured", b, name)
    quit()
    save(b + "-after-quit", stable())
