import json, pathlib, re, subprocess, sys
R = pathlib.Path.cwd()
S = R/'.hive/tmp/t137-check-verifier-2'
E = R/'.hive/spec/golden-frames/tasks/137-gf-frame-chat/evidence/verifier-2'
OLD = E.parent/'verifier'

def run(name, cmd, expected=0, env=None):
    p = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, timeout=240, env=env)
    (E/(name+'.txt')).write_text('$ '+' '.join(map(str,cmd))+'\n'+p.stdout+'\nexit '+str(p.returncode)+'\n')
    print(name, 'exit', p.returncode, flush=True)
    assert (p.returncode == 0) == (expected == 0), (name,p.stdout[-5000:])
    return p

def overlay(name, mapping):
    path = S/(name+'.json')
    path.write_text(json.dumps({'Replace': {str(k):str(v) for k,v in mapping.items()}}))
    return str(path)

if sys.argv[1] == 'boards':
    text = (OLD/'boards.py.txt').read_text().replace('t137-check-verifier', 't137-check-verifier-2').replace('evidence/verifier\'', 'evidence/verifier-2\'')
    (E/'boards.py.txt').write_text(text)
    exec(compile(text, str(E/'boards.py.txt'), 'exec'))
elif sys.argv[1] == 'behavior':
    changed = subprocess.check_output(['git','diff','--name-only','55df4ace...HEAD','--','internal'],text=True).splitlines()
    names = sorted({n for f in changed if f.endswith('_test.go') and (R/f).exists() for n in re.findall(r'^func (Test\w+)\(', (R/f).read_text(), re.M)})
    (E/'behavior-test-names.txt').write_text('\n'.join(names)+'\n')
    regex = '^('+'|'.join(names)+')$'
    run('behavior-all-green', ['go','test','./internal/...','-count=1','-v','-run',regex])
    previous={}
    for f in changed:
        if not f.endswith('.go') or f.endswith('_test.go'): continue
        if not (R/f).exists(): continue
        p=subprocess.run(['git','show','55df4ace:'+f],capture_output=True,text=True)
        dest=S/'before'/f
        dest.parent.mkdir(parents=True,exist_ok=True)
        dest.write_text(p.stdout if p.returncode==0 else 'package '+re.search(r'^package (\w+)',(R/f).read_text(),re.M).group(1)+'\n')
        previous[R/f]=dest
    run('behavior-prechange-red',['go','test','-overlay',overlay('before',previous),'./internal/...','-count=1','-run',regex],expected=1)
    probes={R/'internal/tui/t137_verifier_test.go':OLD/'probes_test.go.txt'}
    run('probes-green',['go','test','-overlay',overlay('probes',probes),'./internal/tui/','-count=1','-v','-run','^TestVerifier'])
    # Reintroduce each earlier failure independently through an overlay.
    for stem,source,test in [('resize','internal/tui/component/composer.go','TestAskResizeKeepsTheTopRightCorner|TestVerifierAskResizeKeepsTheTopRightCorner'),('named-key','internal/lib/home/config.go','TestParseConfigTUIAcceptsNamedKeys|TestVerifierKeysAcceptBubbleTeaNamedKey')]:
        dest=S/(stem+'-before.go')
        dest.write_bytes(subprocess.check_output(['git','show','c4badeaa:'+source]))
        mapping=dict(probes); mapping[R/source]=dest
        run('repair-'+stem+'-red',['go','test','-overlay',overlay(stem+'-before',mapping),'./internal/...','-count=1','-v','-run','^('+test+')$'],expected=1)
        run('repair-'+stem+'-green',['go','test','-overlay',overlay(stem+'-after',probes),'./internal/...','-count=1','-v','-run','^('+test+')$'])
    print('Test names:',len(names),flush=True)
elif sys.argv[1] == 'mutations':
    text=(OLD/'mutations.py.txt').read_text().replace('t137-check-verifier','t137-check-verifier-2').replace('evidence/verifier/mutations','evidence/verifier-2/mutations')
    # Keep the existing, independently written probe source.
    (E/'mutations.py.txt').write_text(text)
    (S/'mutations').mkdir(exist_ok=True)
    cases=[('error','internal/lib/cellgrid/palette.go'),('chord','internal/tui/gframe.go'),('status-lines','internal/tui/gframe.go'),('status-join','internal/tui/component/statusrow.go'),('title','internal/tui/component/titlerow.go'),('throttle','internal/tui/status_command.go'),('demo-command','internal/tui/status_command.go'),('notice','internal/tui/frame.go'),('box','internal/tui/frame.go'),('placeholder','internal/tui/chat.go'),('send','internal/tui/chat.go'),('earlier','internal/tui/chat.go'),('messages','internal/tui/component/chat.go')]
    for name,source in cases:
        subprocess.run(['patch','--silent','-o',str(S/'mutations'/(name+'.go')),str(R/source),str(OLD/'mutations'/(name+'.patch'))],check=True)
    exec(compile(text,str(E/'mutations.py.txt'),'exec'))
elif sys.argv[1] == 'golden':
    run('golden',['go','test','./internal/boardexport/','./internal/tui/','./internal/tui/component/','-count=1','-v','-run','Golden|Gf|Stopped'])
    run('breakpoints',['go','test','./internal/tui/','-count=1','-v','-run','Retired|EachScreenKeepsTheFrame'])
    run('ste',['go','test','./','-run','STE','-count=1'])
elif sys.argv[1] == 'gate':
    commands=['gofmt -l $(git ls-files -co --exclude-standard \'*.go\')','go mod tidy -diff','go vet ./...','golangci-lint run ./...','go test -race ./...']
    output=[]
    for i,cmd in enumerate(commands):
        p=subprocess.run(['bash','-c',cmd],text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,timeout=300)
        output.append('$ '+cmd+'\n'+p.stdout+'exit '+str(p.returncode)+'\n')
        (E/'gate.txt').write_text('\n'.join(output))
        print('gate',i+1,'exit',p.returncode,flush=True)
        assert p.returncode==0 and (i>1 or p.stdout==''), (cmd,p.stdout)
else:
    raise RuntimeError('unknown check')
