import hashlib, json, pathlib, re, subprocess
r = pathlib.Path.cwd()
s = r/'.hive/tmp/t137-check-verifier-2'
e = r/'.hive/spec/golden-frames/tasks/137-gf-frame-chat/evidence/verifier-2'
pinned = pathlib.Path('/home/penguin/source/hive/.hive/tmp/golden-frames-1792795470-106f')
design = r/'.hive/spec/golden-frames/design'
original = s/'original'
original.mkdir(exist_ok=True)
checks = dict(line.split(None,1)[::-1] for line in (pinned/'SHA256SUMS').read_text().splitlines())
files = sorted(design.glob('*.dc.html'))+[design/'canvas.json']
sha = ''
for f in files:
    data = subprocess.check_output(['git','show','6ce218d0:.hive/spec/golden-frames/design/'+f.name])
    (original/f.name).write_bytes(data)
    sha += checks[f.name]+'  '+f.name+'\n'
p = subprocess.run(['sha256sum','-c','-'],input=sha,text=True,capture_output=True,cwd=original)
(e/'board-copy-sha.txt').write_text('$ sha256sum -c - (16 files at copy commit 6ce218d0)\n'+p.stdout+p.stderr+'exit '+str(p.returncode)+'\n')
assert p.returncode == 0
p = subprocess.run(['sha256sum','-c','-'],input=checks['canvas.json']+'  canvas.json\n',text=True,capture_output=True,cwd=design)
with (e/'board-copy-sha.txt').open('a') as out: out.write('Current unchanged file:\n'+p.stdout)
assert p.returncode == 0
for target, source in [(s/'original-grids',original),(s/'export',design)]:
    target.mkdir(exist_ok=True)
    p = subprocess.run(['go','run','./internal/boardexport/cmd/boardexport','-out',str(target),str(source)],text=True,capture_output=True)
    assert p.returncode == 0, p.stderr
report = []
for f in design.glob('*.grid'):
    assert f.read_bytes() == (s/'export'/f.name).read_bytes(), f
    report.append(f.name+': export is unchanged')
p = subprocess.run(['git','status','--short','.hive/spec/golden-frames/design/'],text=True,capture_output=True,check=True)
report += ['$ git status --short .hive/spec/golden-frames/design/',p.stdout or '(no output)']
(e/'export.txt').write_text('\n'.join(report)+'\n')
diffs = e/'board-diffs'
diffs.mkdir(exist_ok=True)
changes = (e.parent/'board-changes.md').read_text()
report = []
for f in design.glob('*.dc.html'):
    old = pinned/'project'/f.name
    p = subprocess.run(['git','diff','--no-index','--',str(old),str(f)],text=True,capture_output=True)
    assert p.returncode in (0,1)
    (diffs/(f.name+'.diff')).write_text(p.stdout)
    stem=f.name.removesuffix('.dc.html')
    before=(s/'original-grids'/(stem+'.grid')).read_text().splitlines()
    after=(design/(stem+'.grid')).read_text().splitlines()
    height=int(before[0].split()[-1])
    rows=[i for i in range(height) if before[1+3*i:4+3*i]!=after[1+3*i:4+3*i]]
    table = next(line for line in changes.splitlines() if line.startswith('| '+stem+' |'))
    listed=[int(x) for x in table.split('|')[3].strip().split(', ')]
    assert rows == listed, (stem,rows,listed)
    oldhash=hashlib.sha256(old.read_bytes()).hexdigest()
    newhash=hashlib.sha256(f.read_bytes()).hexdigest()
    v6 = (e.parent/'V6.md').read_text()
    assert oldhash in v6 and newhash in v6, stem
    report.append(stem+': changed rows match board-changes.md; old/new hashes match V6.md: '+', '.join(map(str,rows)))
(e/'board-changes-check.txt').write_text('\n'.join(report)+'\n')
print('V1: 16 initial hashes OK; current canvas hash OK; 15 exports unchanged')
print('V6: 15 board diff row lists and full hashes agree with the evidence')
