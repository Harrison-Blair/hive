import concurrent.futures, html, pathlib, re, subprocess
ROOT = pathlib.Path.cwd()
SCRATCH = ROOT/'.hive/tmp/t137-check-verifier-2'
E = ROOT/'.hive/spec/golden-frames/tasks/137-gf-frame-chat/evidence/verifier-2'
PAL = {'f':'#e6e1d6','r':'#3d3a33','d':'#a39d90','a':'#f5b942','w':'#6ee69a','o':'#7ccf6a','i':'#6fb7ff','u':'#e9cf9b','c':'#9fd0ff','e':'#e8705f'}
SGR = re.compile(r'\x1b\[([0-9;]*)m')

def grid_rows(path):
    data=path.read_text().splitlines()
    w,h=map(int,data[0].split()[-2:])
    rows=[]
    for y in range(h):
        chars,tokens,bold=[line[2:] for line in data[1+3*y:4+3*y]]
        runs=[]
        for ch,k,b in zip(chars,tokens,bold):
            runs.append('<span style="color:'+PAL[k]+';font-weight:'+('700' if b=='1' else '400')+'">'+html.escape(ch)+'</span>')
        rows.append(''.join(runs))
    return rows

def ansi_rows(path):
    rows=[]
    for line in path.read_text().splitlines()[:40]:
        color=PAL['f']; bold=False; chunks=[]; col=0; pos=0
        for match in list(SGR.finditer(line))+[None]:
            end=match.start() if match else len(line)
            text=line[pos:end][:max(120-col,0)]
            col+=len(text)
            chunks.append('<span style="color:'+color+';font-weight:'+('700' if bold else '400')+'">'+html.escape(text)+'</span>')
            if match is None: break
            codes=[int(x) if x else 0 for x in match.group(1).split(';')]
            i=0
            while i<len(codes):
                c=codes[i]
                if c==0: color=PAL['f']; bold=False
                elif c==1: bold=True
                elif c==22: bold=False
                elif c==39: color=PAL['f']
                elif c==38 and codes[i+1:i+2]==[2] and i+4<len(codes):
                    color='#%02x%02x%02x'%tuple(codes[i+2:i+5]); i+=4
                i+=1
            pos=match.end()
        rows.append(''.join(chunks))
    return rows

def render(job):
    name, rows, out = job
    file=SCRATCH/'html'/(name+'.dc.html')
    file.parent.mkdir(exist_ok=True)
    height=48+17*len(rows)
    file.write_text('<!doctype html><meta charset="utf-8"><style>body{margin:0}pre{margin:0;width:936px;height:'+str(17*len(rows))+'px;padding:24px;background:#171612;color:#e6e1d6;border-radius:10px;font-family:"JetBrains Mono",monospace;font-size:13px;line-height:17px;white-space:pre;overflow:hidden}</style><pre>'+('\n'.join(rows))+'</pre>')
    profile=SCRATCH/'profiles'/name
    profile.mkdir(parents=True, exist_ok=True)
    out.parent.mkdir(parents=True,exist_ok=True)
    cmd=['firefox','--headless','--no-remote','--profile',str(profile),'--window-size=1220,'+str(max(780,height+10)),'--screenshot',str(out),file.as_uri()]
    p=subprocess.run(cmd,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True,timeout=45)
    assert p.returncode==0 and out.exists(), (name,p.stdout)
    return '$ '+' '.join(cmd)+'\n'+p.stdout
jobs=[]
for kind, directory in [('pinned',SCRATCH/'original-grids'),('revised',ROOT/'.hive/spec/golden-frames/design'),('screen',E/'live')]:
    for path in sorted(directory.glob('*.grid')):
        name=kind+'-'+path.stem
        jobs.append((name,grid_rows(path),E/'png'/kind/(path.stem+'.png')))
for path in sorted((E/'live').glob('*.ansi')):
    if '-120x40' not in path.stem:
        jobs.append(('key-'+path.stem,ansi_rows(path),E/'png'/'keys'/(path.stem+'.png')))
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    results=list(pool.map(render,jobs))
(E/'screenshots.txt').write_text('\n'.join(results))
print('Captured',len(jobs),'PNGs with fresh Firefox profiles')
