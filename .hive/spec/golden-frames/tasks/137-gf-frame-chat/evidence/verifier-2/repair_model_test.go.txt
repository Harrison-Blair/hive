package tui

import (
	"os"
	"path/filepath"
	"strings"
	"testing"

	tea "charm.land/bubbletea/v2"
	"github.com/Harrison-Blair/hive/internal/lib/home"
)

// Golden-frames decision 18: the config key must work in the model too.
func TestVerifierNamedKeyOpensHelp(t *testing.T) {
	config := t.TempDir()
	t.Setenv("XDG_CONFIG_HOME", config)
	if err := os.Mkdir(filepath.Join(config, "hive"), 0o755); err != nil {
		t.Fatal(err)
	}
	text := "home = \"/tmp/hive\"\n[tui.keys]\nhelp = \"alt+up\"\n"
	if err := os.WriteFile(filepath.Join(config, "hive", "config.toml"), []byte(text), 0o600); err != nil {
		t.Fatal(err)
	}
	settings, err := home.ReadSettings()
	if err != nil {
		t.Fatal(err)
	}
	m, _ := newFrameModel(t, 120, 40)
	m.setTUIConfig(settings.TUI)
	m, _ = raise(t, m, tea.KeyPressMsg{Code: tea.KeyUp, Mod: tea.ModAlt})
	if !m.helpOpen || !strings.Contains(m.View().Content, "SCREENS") {
		t.Fatal("alt+up did not open Help")
	}
	m, _ = raise(t, m, escKey())
	if m.helpOpen || m.activeTab() != tabChat {
		t.Fatal("esc did not return to Chat")
	}
}
