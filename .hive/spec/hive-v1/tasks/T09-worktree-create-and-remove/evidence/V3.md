```sh
#!/bin/sh
# V3 live transcript probe for T09. It uses one fixture home, one fixture

: > "$FIX/.hive/README.md"
BIN="$FIX/hive-$(date +%s)"

echo "go build exit $?"
sha256sum "$BIN"

echo "git commit exit $?"

echo "project register exit $?"

echo "task create exit $?"
echo '$ herdr workspace list (before)'
herdr workspace list > "$FIX/list-before.json"
cat "$FIX/list-before.json"
echo "workspace list before exit $?"

CREATE_EXIT=$?

cat "$FIX/create.err"
echo "worktree create exit $CREATE_EXIT"
REPO_WS=$(sed -n 's/^repository workspace opened: //p' "$FIX/create.out")

echo "REPO_WS=$REPO_WS WANT_WS=$WANT_WS WT_PATH=$WT_PATH"
echo '$ herdr workspace list (after create)'
herdr workspace list > "$FIX/list-after-create.json"
cat "$FIX/list-after-create.json"
echo "workspace list after create exit $?"
"$BIN" --home "$FIX" worktree remove --task 1-live-demo > "$FIX/remove1.out" 2> "$FIX/remove1.err"
R1=$?
cat "$FIX/remove1.out"
cat "$FIX/remove1.err"
echo "first worktree remove exit $R1"
"$BIN" --home "$FIX" task cancel --task 1-live-demo
echo "task cancel exit $?"
"$BIN" --home "$FIX" worktree remove --task 1-live-demo > "$FIX/remove2.out" 2> "$FIX/remove2.err"
R2=$?
cat "$FIX/remove2.out"
cat "$FIX/remove2.err"
echo "second worktree remove exit $R2"
echo '$ herdr workspace list (after remove)'
herdr workspace list > "$FIX/list-after-remove.json"
cat "$FIX/list-after-remove.json"
echo "workspace list after remove exit $?"
echo '$ git -C "$REPO" branch'
git -C "$REPO" branch
echo "branch list exit $?"
echo '$ git -C "$REPO" worktree list'
git -C "$REPO" worktree list
echo "worktree list exit $?"
echo '[ -e "$WT_PATH" ] check'
if [ -e "$WT_PATH" ]; then echo "worktree path still exists"; else echo "worktree path absent"; fi
if [ -n "$REPO_WS" ]; then
  herdr workspace get "$REPO_WS" > "$FIX/get-repo.json" 2> "$FIX/get-repo.err"
  echo "repo workspace get exit $?"
  cat "$FIX/get-repo.json"
  cat "$FIX/get-repo.err"
  echo '$ herdr workspace close "$REPO_WS"'
  herdr workspace close "$REPO_WS" > "$FIX/close.out" 2> "$FIX/close.err"
  CLOSE_EXIT=$?
  cat "$FIX/close.out"
  cat "$FIX/close.err"
  echo "repository workspace close exit $CLOSE_EXIT"
else
  echo "BLOCKED: no repository workspace ID from create output"
  exit 1
fi
echo '$ herdr workspace list (final)'
herdr workspace list > "$FIX/list-final.json"
cat "$FIX/list-final.json"
echo "workspace list final exit $?"
if grep -q "\"workspace_id\":\"$REPO_WS\"" "$FIX/list-final.json"; then echo "BLOCKED: fixture repository workspace remains"; exit 1; fi
if [ -n "$WANT_WS" ] && grep -q "\"workspace_id\":\"$WANT_WS\"" "$FIX/list-final.json"; then echo "BLOCKED: fixture worktree workspace remains"; exit 1; fi
echo "fixture workspaces absent from final list"
python3 - "$FIX/list-before.json" "$FIX/list-final.json" <<'PY'
import json,sys
before=json.load(open(sys.argv[1]))['result']['workspaces']
final=json.load(open(sys.argv[2]))['result']['workspaces']
before_ids=sorted(w['workspace_id'] for w in before)
final_ids=sorted(w['workspace_id'] for w in final)
print('before ids:', before_ids)
print('final ids: ', final_ids)
print('same ids:', before_ids == final_ids)
raise SystemExit(0 if before_ids == final_ids else 1)
PY
SAME=$?
echo "before/final id comparison exit $SAME"
if [ "$SAME" -ne 0 ]; then echo "BLOCKED: final workspace set differs from before"; exit 1; fi
echo "cleanup: rm -rf $FIX"
rm -rf "$FIX"
echo "fixture cleanup exit $?"
if [ -e "$FIX" ]; then echo "BLOCKED: fixture home remains"; exit 1; fi
echo "fixture home absent"
exit 0
```
