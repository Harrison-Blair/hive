set -u

echo "=== child session files after spawn ==="
find "$SESS" -newer "$MARKER" -type f 2>/dev/null

echo "=== project record after stop ==="

find "$SESS" -newer "$MARKER" -type f 2>/dev/null

herdr workspace list
echo "=== cleanup ==="
rm -rf "$FIX"
echo "removed $FIX"
