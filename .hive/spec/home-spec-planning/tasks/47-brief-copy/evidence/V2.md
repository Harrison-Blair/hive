```sh
    printf '*\n' > "$repo/.gitignore"
    git -C "$repo" add -f README.md .gitignore

python3 -c 'import json,sys; w=json.loads(sys.argv[1])["result"]["workspace"]["worktree"]; assert w["checkout_path"]==sys.argv[2] and not w["is_linked_worktree"]' "$workspace_info" "$repo_allow"

[ "$ignore_exit" -ne 0 ] && [ ! -e "$ignore_wt" ] && [ -z "$(git -C "$repo_ignore" branch --list "task/$ignore_task_id")" ]
```

## Cleanup

The live check removed worktree workspace `wHD` and closed repository workspace `wHC`. Each workspace belonged to the temporary repository. The removal commands returned no output, and the check confirmed that both workspaces were closed.

```sh
for id in wHD wHC; do
  if herdr workspace get "$id" >/dev/null 2>&1; then printf '%s remains\n' "$id"; exit 1; else printf '%s closed\n' "$id"; fi
done
herdr workspace list | python3 -c 'import json,sys; items=json.load(sys.stdin)["result"]["workspaces"]; unexpected=[(x["workspace_id"],(x.get("worktree") or {}).get("checkout_path")) for x in items if ".hive/tmp/47-v2-" in (x.get("worktree") or {}).get("checkout_path", "")]; print("temporary workspaces remaining:", unexpected); raise SystemExit(bool(unexpected))'
```
