import subprocess
from collections import defaultdict
from pathlib import PurePosixPath
base='928cfa95e0906315fcb4a65a5cdbdef1470a0503'
head='8eaa2d8fc4867a57555674d28e99546750468e0a'
rows=subprocess.check_output(['git','diff','--find-renames','--numstat',f'{base}...{head}'],text=True).splitlines()
def category(path):
    p=PurePosixPath(path)
    name=p.name.lower()
    parts={part.lower() for part in p.parts}
    ext=p.suffix.lower()
    if name.endswith(('_test.go','_test.py','_test.js','_test.ts','_test.sh')) or parts & {'test','tests','testdata','fixtures','snapshots'}:
        return 'Tests'
    if ext in {'.md','.markdown','.mdx','.rst','.adoc','.txt'} or name in {'readme','license','copying'}:
        return 'Docs'
    if ext == '.go':
        return 'Code'
    return 'Other'
summary=defaultdict(lambda:[0,0,0])
for row in rows:
    added,deleted,path=row.split('\t',2)
    if added == '-' or deleted == '-':
        raise SystemExit(f'binary stats unavailable: {path}')
    bucket=summary[category(path)]
    bucket[0]+=1; bucket[1]+=int(added); bucket[2]+=int(deleted)
for name in ('Code','Tests','Docs','Other'):
    count,added,deleted=summary[name]
    print(f'{name}: files={count} added={added} deleted={deleted} net={added-deleted}')
print(f'Total: files={sum(v[0] for v in summary.values())} added={sum(v[1] for v in summary.values())} deleted={sum(v[2] for v in summary.values())} net={sum(v[1]-v[2] for v in summary.values())}')
PY
echo "exit $?"
