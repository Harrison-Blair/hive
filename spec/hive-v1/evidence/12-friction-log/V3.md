python3 - <<'PY'
from pathlib import Path
nl = chr(10)
text = Path('reference/dogfood/friction.md').read_text()
def slug(title):
    text = title.replace('`', '').lower()
    text = ''.join(char for char in text if char.isalnum() or char in '_ -')
    return text.strip().replace(' ', '-')
records = {}
for section, stop in [('## Open', '## Resolved'), ('## Resolved', None)]:
    body = text.split(section + nl, 1)[1]
    if stop:
        body = body.split(nl + stop + nl, 1)[0]
    lines = body.splitlines()
    starts = [i for i, line in enumerate(lines) if line.startswith('### ')]
    for i, start in enumerate(starts):
        end = starts[i + 1] if i + 1 < len(starts) else len(lines)
        block = lines[start:end]
        title = block[0][4:]
        issue = next(line[len('**Issue:** '):] for line in block if line.startswith('**Issue:** '))
        status = next(line[len('**Status:** '):] for line in block if line.startswith('**Status:** '))
        assert title == issue
        records[title] = status
rows = []
for line in text.splitlines():
    if line.startswith('| [') and line.endswith('|'):
        cells = [cell.strip() for cell in line.split('|')]
        link, status = cells[1], cells[2]
        title = link[1:link.index(']')]
        target = link.split('(#', 1)[1][:-1]
        assert target == slug(title), (title, target, slug(title))
        assert title in records and records[title] == status
        rows.append(title)
assert rows == list(records)
print(f'index links={len(rows)} headings={len(records)}')
print('all links target matching headings; all row statuses match entries')
PY
echo "exit $?"
